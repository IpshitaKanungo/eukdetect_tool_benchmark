2) Metrics & confusion matrix helpers¶

We compute TP/TN/FP/FN and the following metrics per tool:

  • Precision = TP / (TP+FP) (Of all contigs the tool called “euk”, how many were actually euk?)
  • Recall (Sensitivity) = TP / (TP+FN) (Of all true euks, how many did the tool find?)
  • Specificity (TNR) = TN / (TN+FP) (Of all true non-euks, how many did the tool correctly reject?)
In [26]:
pd.set_option('display.float_format', '{:.4f}'.format)
def confusion_counts(y_true: pd.Series, y_pred: pd.Series):
    TP = ((y_true == True) & (y_pred == True)).sum()
    TN = ((y_true == False) & (y_pred == False)).sum()
    FP = ((y_true == False) & (y_pred == True)).sum()
    FN = ((y_true == True) & (y_pred == False)).sum()
    return TP, TN, FP, FN

def _safe_div(num, den):
    return np.nan if den == 0 else (num / den)
    
def compute_metrics_df(subdf: pd.DataFrame, tool_col: str) -> pd.Series:
    y_true = subdf['ground_truth']
    y_pred = subdf[tool_col]

    TP, TN, FP, FN = confusion_counts(y_true, y_pred)

    precision = _safe_div(TP, TP + FP)
    recall    = _safe_div(TP, TP + FN)
    specificity = _safe_div(TN, TN + FP)
    if precision == 0:
        precision = np.nan
    if recall == 0:
        recall = np.nan
    if specificity == 0:
        specificity = np.nan

    GT_true = (y_true == True).sum()
    GT_false = (y_true == False).sum()

    out = pd.Series({
        'Ground Truth True': GT_true,
        'Ground Truth False': GT_false,
        'TP': TP, 'TN': TN, 'FP': FP, 'FN': FN,
        'Precision': precision, 'Recall': recall, 'Specificity': specificity,
    })
    return out

def metrics_table_all_tools(df_in: pd.DataFrame, tool_cols):
    rows = []
    for tool in tool_cols:
        s = compute_metrics_df(df_in, tool)
        s.name = tool
        rows.append(s)
    return pd.DataFrame(rows)

5) Base-adjusted metrics (weighted by contig length)¶

Sometimes assemblies care about bp rather than contig counts.
We compute length-weighted confusion counts:

  • TP_bp = sum(length) over contigs where (GT=True & Pred=True)
  • TN_bp, FP_bp, FN_bp similarly

Then recompute metrics with bp-weighted counts.

In [35]:
def compute_metrics_weighted_bp(df_in: pd.DataFrame, tool_col: str) -> pd.Series:
    y_true = df_in['ground_truth'].astype(bool)
    y_pred = df_in[tool_col].astype(bool)
    L      = pd.to_numeric(df_in['length'], errors='coerce').fillna(0)

    # bp-weighted confusion "counts"
    TP_bp = L[(y_true)  & (y_pred)].sum()
    TN_bp = L[(~y_true) & (~y_pred)].sum()
    FP_bp = L[(~y_true) & (y_pred)].sum()
    FN_bp = L[(y_true)  & (~y_pred)].sum()

    GT_true_bp  = L[y_true].sum()
    GT_false_bp = L[~y_true].sum()

    # length-weighted metrics
    precision = TP_bp / (TP_bp + FP_bp) if (TP_bp + FP_bp) > 0 else np.nan
    recall    = TP_bp / (TP_bp + FN_bp) if (TP_bp + FN_bp) > 0 else np.nan
    specificity = TN_bp / (TN_bp + FP_bp) if (TN_bp + FP_bp) > 0 else np.nan

    # --- numerically stable MCC using scaled counts (prevents overflow)
    # scale by total so all terms ∈ [0,1]
    total_bp = TP_bp + TN_bp + FP_bp + FN_bp
    if total_bp > 0:
        tp = TP_bp / total_bp
        tn = TN_bp / total_bp
        fp = FP_bp / total_bp
        fn = FN_bp / total_bp

        num = tp*tn - fp*fn
    # ---------------------------------------------

    return pd.Series({
        'Ground Truth True': GT_true_bp,
        'Ground Truth False': GT_false_bp,
        'TP': TP_bp, 'TN': TN_bp, 'FP': FP_bp, 'FN': FN_bp,
        'Precision': precision, 'Recall': recall, 'Specificity': specificity,
    })

def metrics_bp_table_all_tools(df_in: pd.DataFrame, tool_cols):
    rows = []
    for tool in tool_cols:
        s = compute_metrics_weighted_bp(df_in, tool)
        s.name = tool
        rows.append(s)
    return pd.DataFrame(rows)

def plot_confusion_from_df(metrics_df, tool_name, normalize='total'):
    """
    Confusion matrix for a single tool from bp-weighted metrics table.

    metrics_df: DataFrame with index=tool, columns [TP_bp, TN_bp, FP_bp, FN_bp]
    tool_name:  which tool (row index in the df)
    normalize:  'total' -> color by fraction of all samples
                'row'   -> color by row
                'col'   -> color by column
                None    -> raw counts for color intensity
    """
    row = metrics_df.loc[tool_name]
    TP, TN, FP, FN = row["TP"], row["TN"], row["FP"], row["FN"]

    # Arrange like schema: rows = predicted, cols = true
    M = np.array([[TP, FP],
                  [FN, TN]], dtype=float)

    # normalization for color
    C = M.copy()
    if normalize == 'total':
        tot = C.sum()
        C = C / tot if tot > 0 else C
    elif normalize == 'row':
        row_sums = C.sum(axis=1, keepdims=True)
        C = np.divide(C, row_sums, out=np.zeros_like(C), where=row_sums!=0)
    elif normalize == 'col':
        col_sums = C.sum(axis=0, keepdims=True)
        C = np.divide(C, col_sums, out=np.zeros_like(C), where=col_sums!=0)

    fig, ax = plt.subplots(figsize=(5,4))
    im = ax.imshow(C, cmap='Blues', vmin=0, vmax=C.max() if C.max()>0 else 1)

    # gridlines
    for x in (0.5, 1.5):
        ax.axvline(x, color='white', lw=2)
        ax.axhline(x, color='white', lw=2)

    # ticks
    ax.set_xticks([0,1]); ax.set_yticks([0,1])
    ax.set_xticklabels(['Positive', 'Negative'])
    ax.set_yticklabels(['Positive', 'Negative'])
    ax.set_xlabel('True Class')
    ax.set_ylabel('Predicted Class')
    ax.set_title(f'Confusion Matrix — {tool_name}')

    # annotations
    total = M.sum()
    labels = np.array([['True Positive', 'False Positive'],
                       ['False Negative', 'True Negative']])
    for i in range(2):
        for j in range(2):
            count = M[i,j]
            frac  = (count / total) if total>0 else 0
            txt = f"{labels[i,j]}\n{count:,.0f}\n({frac*100:.2f}%)"
            ax.text(j, i, txt, ha='center', va='center',
                    color='white' if C[i,j] > (C.max()*0.5) else 'black',
                    fontsize=9, fontweight='bold')

    plt.tight_layout()
    plt.show()

def assign_quality_tier(row):
    cpl, cnt = row['completeness'], row['contamination']
    # Example categories (adjust to your study's definitions)
    if cpl >= 90 and cnt <= 5:
        return 'Almost Complete'
    elif cpl >= 70 and cnt <= 5:
        return 'High Quality'
    elif cpl >= 50 and cnt <= 10:
        return 'Medium Quality'
    else:
        return 'Other' # shouldn't be the case of yours, because all bins are >50% in completeness; still good to have for a sanity check
