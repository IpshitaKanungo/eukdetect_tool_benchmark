import re
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# ---------- Combine your data ----------
df_15_ = df_15.copy(); df_15_["contam"] = 15
df_50_ = df_50.copy(); df_50_["contam"] = 50
df_85_ = df_85.copy(); df_85_["contam"] = 85
df_all = pd.concat([df_15_, df_50_, df_85_], ignore_index=True)

# Ensure numeric types
for c in ["TP","TN","FP","FN","Precision","Recall","Specificity","F1"
          "Ground Truth True","Ground Truth False"]:
    if c in df_all.columns:
        df_all[c] = pd.to_numeric(df_all[c], errors="coerce")

TOOLS = ["tiara","kraken2","eukrep"]

# Distinct colors per tool (stay consistent across figures)
tool_color = {
    "tiara":   "#D62728",  # red
    "kraken2": "#1F77B4",  # blue
    "eukrep":  "#2CA02C",  # green
}
# Line style encodes dataset (host %)
linestyle = {15: ":", 50: "--", 85: "-"}

marker = {"tiara":"o", "kraken2":"s", "eukrep":"^"}

def order_bins(labels):
    def lb(s):
        s = str(s)
        if "≥" in s:
            m = re.findall(r"\d+", s)
            return int(m[0]) if m else 10**12
        m = re.findall(r"\d+", s)
        return int(m[0]) if m else -1
    return sorted(set(map(str, labels)), key=lb)

# ---- Simplified version: only the top metric plot ----
def plot_metric_lines(df_all, metric):
    """
    Creates a clean line+dot plot for the given metric
    (Precision, Recall, or Specificity) across tools and datasets.
    """
    bins = order_bins(df_all["length_bin"].unique())
    x = np.arange(len(bins))

    fig, ax = plt.subplots(figsize=(8, 5))
    for tool in TOOLS:
        for lv in [15, 50, 85]:
            sub = df_all[(df_all["Tool"] == tool) & (df_all["contam"] == lv)].set_index("length_bin")
            y = [sub.loc[b, metric] if (b in sub.index and pd.notna(sub.loc[b, metric])) else np.nan
                 for b in bins]
            ax.plot(
                x, y,
                color=tool_color[tool],
                linestyle=linestyle[lv],
                marker=marker[tool],
                alpha=0.45,
                label=f"{tool} {lv}%"
            )

    ax.set_xlim(-0.5, len(bins) - 0.5)
    ax.set_ylim(0, 1.02)
    ax.set_ylabel(metric)
    ax.set_title(metric)
    ax.grid(alpha=0.25, axis="y")
    ax.set_xticks(x)
    ax.set_xticklabels(bins, rotation=25)
    ax.legend(ncol=3, frameon=False, loc="lower right")

    fig.suptitle(f"{metric} across tools (color=tool, linestyle=host%)", y=0.98)
    fig.tight_layout()
    return fig

# ---- Build the three figures ----
fig_prec = plot_metric_lines(df_all, "Precision")
fig_rec  = plot_metric_lines(df_all, "Recall")
fig_spec = plot_metric_lines(df_all, "Specificity")
fig_spec = plot_metric_lines(df_all, "F1")
plt.show()

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from pathlib import Path

from matplotlib.patches import Patch

# bins.tsv obtained from data post-processing binning step
df_15bins = pd.read_csv("15_bins.tsv", sep="\t")
df_50bins = pd.read_csv("50_bins.tsv", sep="\t")
df_85bins = pd.read_csv("85_bins.tsv", sep="\t")

RENAME = {
    "dirty_metagenome": "contaminated metagenome",
    "metagenome": "original metagenome",
}

def summarize_from_quality_table(df: pd.DataFrame) -> pd.DataFrame:
    d = df.copy()

    # Apply readable dataset names
    if RENAME:
        d['dataset'] = d['dataset'].replace(RENAME)
    #d['dataset'] = d['dataset'].replace(RENAME)

    # Normalise quality text
    d['quality_norm'] = (
        d['quality'].astype(str)
        .str.strip()
        .str.replace(r'\s+', ' ', regex=True)
        .str.lower()
    )

    # counts are in the 'bin' column (rename if your file uses a different name)
    count_col = 'bin'
    if count_col not in d.columns:
        raise ValueError("Expected a 'bin' column with counts.")

    # total bins per dataset
    total = d.groupby('dataset', as_index=True)[count_col].sum().rename('total_bins')

    # "Almost Complete" per dataset (case-insensitive)
    ac = (d.loc[d['quality_norm'].eq('almost complete')]
            .groupby('dataset')[count_col]
            .sum()
            .reindex(total.index, fill_value=0)
            .rename('almost_complete'))

    out = (
        pd.concat([total, ac], axis=1)
          .reset_index()
          .sort_values('dataset')
    )
    return out

sum15 = summarize_from_quality_table(df_15bins); sum15["contam"] = 15
sum50 = summarize_from_quality_table(df_50bins); sum50["contam"] = 50
sum85 = summarize_from_quality_table(df_85bins); sum85["contam"] = 85
sum_all = pd.concat([sum15, sum50, sum85], ignore_index=True)


DATASET_ORDER = [
    "bowtie2",
    "original metagenome",
    "contaminated metagenome",
    "eukrep",
    "kraken2",
    "tiara",
]
COL = {15: "#D62728", 50: "#1f77b4", 85: "#2ca02c"}  # contamination colors

def plot_grouped_stacked_counts(summary_all, title="Bins per dataset — all contamination levels"):
    # keep only datasets that exist, in desired order
    datasets = [d for d in DATASET_ORDER if d in summary_all["dataset"].unique()]
    x = np.arange(len(datasets))
    width = 0.22

    fig, ax = plt.subplots(figsize=(10, 5))
    ymax = 0

    for i, lv in enumerate([15, 50, 85]):
        sub = (summary_all[summary_all["contam"] == lv]
               .set_index("dataset")
               .reindex(datasets))
        total = sub["total_bins"].to_numpy()
        ac    = sub["almost_complete"].to_numpy()
        other = np.clip(total - ac, 0, None)

        # position for this contamination's bars
        xi = x + (i-1)*width

        # bottom segment = Almost complete
        ac_bar = ax.bar(xi, ac, width, label=f"{lv}%", color=COL[lv], alpha=1.0)

        try:
            ax.bar_label(
                ac_bar,
                labels=[f"{int(v)}" for v in ac],
                label_type="center",
                fontsize=9,
                color="white",
                fontweight="bold"
            )
        except AttributeError:
            # fallback for older Matplotlib
            for rect, v in zip(ac_bar, ac):
                h = rect.get_height()
                # switch to black text if segment is tiny
                txt_color = "white" if h >= 3 else "black"
                ax.text(rect.get_x() + rect.get_width()/2,
                        rect.get_y() + h/2,
                        f"{int(v)}",
                        ha="center", va="center",
                        fontsize=9, color=txt_color, fontweight="bold")

        # top segment = the rest (High + Medium etc.)
        other_bar = ax.bar(xi, other, width, bottom=ac,
                           color=COL[lv], alpha=0.35)

        # value label at the top of total
        for r, htot in zip(ac_bar, total):
            ax.text(r.get_x() + r.get_width()/2, htot + 0.6, f"{int(htot)}",
                    ha="center", va="bottom", fontsize=9)

        ymax = max(ymax, (total.max() if len(total) else 0))

    ax.set_xticks(x)
    ax.set_xticklabels(datasets, rotation=20, ha="right")
    ax.set_xlabel("Dataset")
    ax.set_ylabel("Number of bins")
    ax.set_title(title)
    ax.set_ylim(0, ymax * 1.20)

    # Two-part legend: contamination colors + stack meaning
    contam_handles = [Patch(facecolor=COL[c], label=f"{c}%") for c in [15, 50, 85]]
    stack_handles  = [Patch(facecolor="gray", alpha=0.35, label="Total MAGs"),
                      Patch(facecolor="gray", alpha=1.0,  label="Almost complete")]
    leg_stack = ax.legend(handles=stack_handles,
                          loc="center left", bbox_to_anchor=(1.02, 0.65), frameon=False)
    ax.add_artist(leg_stack)
    ax.legend(handles=contam_handles, title="Contamination level",
              loc="center left", bbox_to_anchor=(1.02, 0.25), frameon=False)
    plt.subplots_adjust(right=0.80)
    plt.tight_layout()
    return fig

plot_grouped_stacked_counts(sum_all)
plt.show()
